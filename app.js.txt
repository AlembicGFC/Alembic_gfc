function isBirthday(dob) {

  const d = new Date(dob);
  const today = new Date();
  return d.getDate() === today.getDate() && d.getMonth() === today.getMonth();
}

function handlePhotoUpload(event, urlInputId, previewId) {
  const file = event.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = function(e) {
    const img = new Image();
    img.onload = function() {
      const canvas = document.createElement('canvas');
      let width = img.width, height = img.height;
      const max_size = 250;
      if (width > height) { if (width > max_size) { height *= max_size / width; width = max_size; } }
      else { if (height > max_size) { width *= max_size / height; height = max_size; } }
      canvas.width = width; canvas.height = height;
      const ctx = canvas.getContext('2d');
      ctx.drawImage(img, 0, 0, width, height);
      const dataUrl = canvas.toDataURL('image/jpeg', 0.8);
      document.getElementById(urlInputId).value = dataUrl;
      const preview = document.getElementById(previewId);
      preview.src = dataUrl; preview.style.display = 'block';
    };
    img.src = e.target.result;
  };
  reader.readAsDataURL(file);
}

// ✅ Perfect Pediatric BMI Calculation (Age & Gender Wise)
function getPediatricBMI(age, gender, heightCm, weightKg) {
  if(!heightCm || !weightKg) return { bmi: null, status: 'No Data', color: '#6b7280' };
  
  const hM = heightCm / 100;
  const bmi = weightKg / (hM * hM);
  
  const thresholds = {
    5: { boy: [13.4, 16.6, 18.2], girl: [13.1, 16.7, 18.2] },
    6: { boy: [13.4, 16.9, 18.5], girl: [13.1, 17.0, 18.8] },
    7: { boy: [13.5, 17.3, 19.1], girl: [13.1, 17.4, 19.6] },
    8: { boy: [13.7, 17.9, 20.0], girl: [13.3, 18.0, 20.6] },
    9: { boy: [14.0, 18.6, 21.0], girl: [13.5, 18.6, 21.6] },
    10: { boy: [14.4, 19.4, 22.1], girl: [13.7, 19.4, 22.7] },
    11: { boy: [14.8, 20.2, 23.2], girl: [14.1, 20.2, 23.7] },
    12: { boy: [15.4, 21.0, 24.4], girl: [14.6, 21.1, 24.8] },
    13: { boy: [16.0, 21.9, 25.6], girl: [15.2, 22.0, 25.9] },
    14: { boy: [16.7, 22.7, 26.8], girl: [15.8, 22.8, 26.9] },
    15: { boy: [17.4, 23.5, 27.9], girl: [16.4, 23.5, 27.9] },
    16: { boy: [18.2, 24.3, 28.9], girl: [17.0, 24.1, 28.8] },
    17: { boy: [18.9, 25.0, 29.8], girl: [17.5, 24.6, 29.5] },
    18: { boy: [19.5, 25.6, 30.5], girl: [18.0, 25.0, 30.1] }
  };

  let ageInt = parseInt(age) || 0;
  if (ageInt < 5) ageInt = 5; 
  if (ageInt > 18) ageInt = 18;
  
  const genderKey = (gender && gender.toLowerCase().startsWith('f')) ? 'girl' : 'boy';
  const limits = thresholds[ageInt][genderKey];
  
  let status = 'Normal';
  let color = '#16a34a';
  
  if (bmi < limits[0]) { status = 'Underweight'; color = '#92400e'; }
  else if (bmi > limits[2]) { status = 'Obese'; color = '#991b1b'; }
  else if (bmi > limits[1]) { status = 'Overweight'; color = '#9a3412'; }
  
  return { bmi: bmi.toFixed(1), status, color };
}

async function loadBarCharts(){
  const fromVal = document.getElementById('reportFromDate').value;
  const toVal = document.getElementById('reportToDate').value;

  const {data: master} = await sb.from('kra_kpi_master').select('*').order('kpi_code');
  
  let dailyQuery = sb.from('kpi_daily').select('*');
  if(fromVal && toVal){
    dailyQuery = dailyQuery.gte('entry_date', fromVal).lte('entry_date', toVal);
  }
  const {data: daily} = await dailyQuery;

  const kpiAverages = {};
  (daily||[]).forEach(d => {
      if(!kpiAverages[d.kpi_code]) kpiAverages[d.kpi_code] = { sum: 0, count: 0 };
      if(d.rating != null) {
          kpiAverages[d.kpi_code].sum += parseFloat(d.rating) * 10;
          kpiAverages[d.kpi_code].count++;
      }
  });

  const kras = {};
  (master||[]).forEach(k=>{
    if(!kras[k.kra_code]) kras[k.kra_code] = {code:k.kra_code, name:k.kra_name, kpis:[]};
    kras[k.kra_code].kpis.push(k);
  });

  const grid = document.getElementById('barChartsGrid');
  grid.innerHTML = ''; 

  Object.values(barCharts).forEach(c => c.destroy());
  barCharts = {};

  Object.keys(kras).sort().forEach(code => {
    const k = kras[code];
    
    const div = document.createElement('div');
    div.className = 'chart-card';
    // Onclick to expand
    div.onclick = () => openChartModal(code, kras[code], kpiAverages);
    
    div.innerHTML = `<h4 title="${k.name}">${code} — ${k.name}</h4><div class="chart-wrap-sm"><canvas id="chart-${code}"></canvas></div>`;
    grid.appendChild(div);

    const labels = k.kpis.map(x => x.kpi_code);
    const dataValues = k.kpis.map(x => {
      const avgObj = kpiAverages[x.kpi_code];
      return avgObj && avgObj.count > 0 ? Math.round(avgObj.sum / avgObj.count) : 0;
    });
    
    const bgColors = dataValues.map(v => v >= 85 ? '#16a34a' : v >= 60 ? '#f59e0b' : '#dc2626');
    const tooltips = k.kpis.map(x => x.kpi_name);

    const ctx = document.getElementById(`chart-${code}`).getContext('2d');
    barCharts[code] = new Chart(ctx, {
      type: 'bar',
      data: {
        labels: labels,
        datasets: [{ label: 'Score %', data: dataValues, backgroundColor: bgColors, borderRadius: 3 }]
      },
      options: {
        responsive: true,
        maintainAspectRatio: false,
        plugins: { 
          legend: { display: false },
          tooltip: { callbacks: { title: (context) => tooltips[context[0].dataIndex] } }
        },
        scales: {
          y: { beginAtZero: true, max: 100, display: false },
          x: { ticks: { font: { size: 8 } } }
        }
      }
    });
  });
}

function showKraDetails(kraCode) {
  const kra = kraCache[kraCode];
  if(!kra) return;
  
  document.getElementById('profileModalTitle').textContent = '🎯 ' + kra.code + ' — ' + kra.name + ' (Details)';
  const body = document.getElementById('profileBody');
  
  let html = `<div class="profile-header-bg">
      <div>
        <div style="font-size:1.3rem;font-weight:800">${kra.code} — ${escapeHTML(kra.name)}</div>
        <div style="font-size:.85rem;color:#475569">Priority: P${kra.priority} • ${kra.kpis.length} KPIs tracked</div>
      </div>
    </div>`;
    
  html += `<table class="kpi-table"><thead><tr><th>Code</th><th>KPI Name</th><th>Latest Score</th><th>Status</th></tr></thead><tbody>`;
  
  kra.kpis.forEach(k => {
    const latest = kra.latest[k.kpi_code];
    const val = latest ? parseFloat(latest.rating) * 10 : null;
    const pillCls = val !== null ? (val >= 85 ? 'pill-good' : val >= 60 ? 'pill-warn' : 'pill-bad') : 'pill-nodata';
    const statusText = val !== null ? (val >= 85 ? '🟢 Good' : val >= 60 ? '🟡 Average' : '🔴 Poor') : '⚪ No Data';
    const displayVal = val !== null ? Math.round(val) + '%' : '—';
    
    html += `<tr>
      <td class="col-code">${k.kpi_code}</td>
      <td class="col-name">${escapeHTML(k.kpi_name)}</td>
      <td class="col-num"><span class="kpi-score-pill ${pillCls}">${displayVal}</span></td>
      <td>${statusText}</td>
    </tr>`;
  });
  
  html += `</tbody></table>`;
  
  body.innerHTML = html;
  document.getElementById('profileModal').classList.add('show');
}

function getRatingLabel(r){
  const labels = {
    '1':'❌ Critical (10%)','2':'❌ Very Poor (20%)','3':'⚠ Poor (30%)','4':'⚠ Below Avg (40%)',
    '5':'➖ Average (50%)','6':'👍 Above Avg (60%)','7':'👍 Good (70%)',
    '8':'✅ Very Good (80%)','9':'⭐ Excellent (90%)','10':'⭐ Excellent (100%)'
  };
  return labels[String(r)] || '';
}

function renderKpiRatingButtons(){

async function loadKraCards(){
  const fromVal = document.getElementById('reportFromDate').value;
  const toVal = document.getElementById('reportToDate').value;

  const {data:master} = await sb.from('kra_kpi_master').select('*').order('kpi_code');
  
  let dailyQuery = sb.from('kpi_daily').select('*');
  if(fromVal && toVal){
    dailyQuery = dailyQuery.gte('entry_date', fromVal).lte('entry_date', toVal);
  }
  const {data:daily} = await dailyQuery;
  
  kpiMasterCache = master || [];

  const kras = {};
  (master||[]).forEach(k=>{
    if(!kras[k.kra_code]) kras[k.kra_code] = {code:k.kra_code,name:k.kra_name,priority:k.priority,kpis:[]};
    kras[k.kra_code].kpis.push(k);
  });

  const kpiAverages = {};
  (daily||[]).forEach(d => {
      if(!kpiAverages[d.kpi_code]) kpiAverages[d.kpi_code] = { sum: 0, count: 0 };
      if(d.rating != null) {
          kpiAverages[d.kpi_code].sum += parseFloat(d.rating) * 10;
          kpiAverages[d.kpi_code].count++;
      }
  });

  Object.values(kras).forEach(kra=>{
    let sum=0,count=0;
    kra.kpis.forEach(k=>{
      const avgObj = kpiAverages[k.kpi_code];
      if(avgObj && avgObj.count > 0){
        sum += (avgObj.sum / avgObj.count);
        count++;
      }
    });
    kra.score = count>0 ? Math.round(sum/count) : null;
    kra.latest = {};
    kra.kpis.forEach(k => {
        const avgObj = kpiAverages[k.kpi_code];
        if(avgObj) kra.latest[k.kpi_code] = { rating: (avgObj.sum / avgObj.count) / 10 };
    });
  });

  kraCache = kras;
  const grid = document.getElementById('kraGrid');
  if(!grid) return;
  
  grid.innerHTML = Object.keys(kras).sort().map(code=>{
    const k = kras[code];
    const score = k.score != null ? k.score : 0;
    const color = score >= 85 ? '#16a34a' : score >= 60 ? '#f59e0b' : '#dc2626';
    return `<div class="kra-card" onclick="showKraDetails('${code}')">
        <div class="kra-code">${code} · P${k.priority}</div>
        <div class="kra-name">${escapeHTML(k.name)}</div>
        <div class="kra-score" style="color:${color}">${k.score!=null ? k.score+'%' : '—'}</div>
        <div class="kra-bar-bg"><div class="kra-bar-fill" style="width:${score}%;background:${color}"></div></div>
        <div class="kra-sub">${k.kpis.length} KPIs tracked</div>
      </div>`;
  }).join('');

  const sel = document.getElementById('kpiSelect');
  if(sel) sel.innerHTML = '<option value="">-- Select KPI --</option>' + kpiMasterCache.map(k=>
    `<option value="${k.kpi_code}">${k.kpi_code} — ${escapeHTML(k.kpi_name)} [${escapeHTML(k.kra_code)}]</option>`
  ).join('');
}

function openPrintModal() {
  document.getElementById('printModal').classList.add('show');
}
function closePrintModal() {
  document.getElementById('printModal').classList.remove('show');
}

function executePrint() {
  const printArea = document.getElementById('printArea');
  printArea.innerHTML = ''; 

  const selectedKras = Array.from(document.querySelectorAll('.print-kra:checked')).map(cb => cb.value);
  const childOption = document.querySelector('input[name="printChild"]:checked').value;
  const staffOption = document.querySelector('input[name="printStaff"]:checked').value;

  let html = `<div class="print-title">Alembic GFC Report</div>`;
  html += `<p style="text-align:center; margin-bottom: 20px;">Date: ${new Date().toLocaleDateString()}</p>`;

  if (selectedKras.length > 0) {
    html += `<h3 style="margin-bottom: 15px; border-bottom: 1px solid #ccc; padding-bottom: 5px;">KRA Performance Overview</h3>`;
    html += `<div style="display: flex; flex-wrap: wrap; gap: 15px; margin-bottom: 20px;">`;
    selectedKras.forEach(code => {
      const k = kraCache[code];
      if(k) {
        const score = k.score != null ? k.score : 0;
        html += `
          <div class="print-kpi-item" style="border: 1px solid #000; padding: 10px; border-radius: 8px; min-width: 150px;">
            <div style="font-size: 10px; font-weight: bold; color: #555;">${k.code} - P${k.priority}</div>
            <div style="font-size: 14px; font-weight: bold; margin: 5px 0;">${escapeHTML(k.name)}</div>
            <div style="font-size: 20px; font-weight: bold;">${score}%</div>
            <div style="font-size: 10px;">${k.kpis.length} KPIs tracked</div>
          </div>
        `;
      }
    });
    html += `</div>`;
  }

  if (childOption !== 'none') {
    const active = allChildrenCache.filter(c=>c.status==='active');
    html += `<h3 style="margin-top: 30px; margin-bottom: 10px; border-bottom: 1px solid #ccc; padding-bottom: 5px;">Children List</h3>`;
    html += `<table class="print-table"><thead><tr>`;
    
    if (childOption === 'full') {
      html += `<th>Sr</th><th>GFC ID</th><th>Name</th><th>Age</th><th>Gender</th><th>Guardian</th><th>School</th><th>Status</th>`;
    } else {
      html += `<th>Sr</th><th>Name</th><th>Age</th>`;
    }
    html += `</tr></thead><tbody>`;

    active.forEach((c, index) => {
      html += `<tr>`;
      if (childOption === 'full') {
        html += `<td>${index+1}</td><td>${escapeHTML(c.gfc_id||'–')}</td><td>${escapeHTML(c.full_name)}</td><td>${calcAge(c.dob)||'–'}</td><td>${escapeHTML(c.gender||'–')}</td><td>${escapeHTML(c.guardian_name||'–')}</td><td>${escapeHTML(c.present_school||'–')}</td><td>${c.status.toUpperCase()}</td>`;
      } else {
        html += `<td>${index+1}</td><td>${escapeHTML(c.full_name)}</td><td>${calcAge(c.dob)||'–'}</td>`;
      }
      html += `</tr>`;
    });
    html += `</tbody></table>`;
  }

  if (staffOption !== 'none') {
    const activeStaff = allStaffCache.filter(s => s.status === 'active');
    html += `<h3 style="margin-top: 30px; margin-bottom: 10px; border-bottom: 1px solid #ccc; padding-bottom: 5px;">Staff List</h3>`;
    html += `<table class="print-table"><thead><tr><th>Sr</th><th>Name</th><th>Role</th><th>Shift</th><th>Contact</th></tr></thead><tbody>`;
    activeStaff.forEach((s, index) => {
      html += `<tr><td>${index+1}</td><td>${escapeHTML(s.full_name)}</td><td>${escapeHTML(s.role||'–')}</td><td>${escapeHTML(s.shift_timing||'–')}</td><td>${escapeHTML(s.contact_no||'–')}</td></tr>`;
    });
    html += `</tbody></table>`;
  }

  printArea.innerHTML = html;
  closePrintModal();

  setTimeout(() => {
    window.print();
  }, 300);
}
